const BACKEND_URL = "https://magic-bot-backend-ym8f.onrender.com"; async function sendOrder(orderData) { try { const response = await fetch(`${BACKEND_URL}/api/order`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(orderData) }); const result = await response.json(); if (response.ok && result.status === 'ok') { Telegram.WebApp.close(); // Закрываем WebApp после успешной отправки } else { alert(result.message || "Произошла ошибка при отправке заявки."); } } catch (error) { console.error("Ошибка сети:", error); alert("Не удалось связаться с сервером."); } }
